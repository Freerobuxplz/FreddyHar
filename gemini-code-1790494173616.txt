-- Dropper System
local function StartDropper(model)
	local dropPoint = model:FindFirstChild("DropPoint")
	local dropValue = model:GetAttribute("DropValue") or 5
	local dropRate = model:GetAttribute("DropRate") or 2

	if not dropPoint then return end
	
	coroutine.wrap(function()
		while model.Parent == Unlocked do
			local ore = Instance.new("Part")
			ore.Size = Vector3.new(1, 1, 1)
			ore.Position = dropPoint.Position
			ore.BrickColor = BrickColor.new("Bright yellow")
			ore.Material = Enum.Material.Neon
			ore:SetAttribute("CashValue", dropValue)
			ore.Parent = workspace
			
			-- Clean up ores after 15s to prevent lag
			game.Debris:AddItem(ore, 15)
			task.wait(dropRate)
		end
	end)()
end

-- Button Purchase System
local function SetupButton(btn)
	local cost = btn:GetAttribute("Cost") or 10
	local targetName = btn:GetAttribute("Target")
	local db = false

	-- Auto-generate Floating UI for cost
	local bGui = Instance.new("BillboardGui", btn)
	bGui.Size = UDim2.new(0, 100, 0, 50)
	bGui.StudsOffset = Vector3.new(0, 2, 0)
	local bTxt = Instance.new("TextLabel", bGui)
	bTxt.Size = UDim2.new(1,0,1,0)
	bTxt.BackgroundTransparency = 1
	bTxt.TextScaled = true
	bTxt.Font = Enum.Font.GothamBold
	bTxt.TextColor3 = Color3.new(0.3, 1, 0.3)
	bTxt.Text = "$" .. tostring(cost)
	
	-- Special coloring for free buttons
	if cost == 0 then
		bTxt.Text = "FREE"
		btn.BrickColor = BrickColor.new("Lime green")
	end

	btn.Touched:Connect(function(hit)
		if db or not Owner then return end
		local player = game.Players:GetPlayerFromCharacter(hit.Parent)
		
		if player == Owner then
			db = true
			local ls = player:FindFirstChild("leaderstats")
			
			if ls and ls.Cash.Value >= cost then
				local targetItem = Purchasables:FindFirstChild(targetName)
				if targetItem then
					ls.Cash.Value -= cost
					targetItem.Parent = Unlocked
					
					-- If it's a dropper, activate its drop loop
					if targetItem:GetAttribute("IsDropper") then
						StartDropper(targetItem)
					end
					
					btn:Destroy()
				end
			end
			task.wait(0.5)
			db = false
		end
	end)
end

-- Initialize Buttons

end
